## Q31. Longest Customer Purchase Streak

**Question:** Find the customer who has the longest sequence of consecutive purchase days.

**Answer:**

```sql
WITH distinct_segment AS (
 SELECT DISTINCT
 c.customer_unique_id,
 CAST (o.order_purchase_timestamp AS DATE) AS purchase_date
 FROM customers c
 JOIN orders o
 ON c.customer_id = o.customer_id
),
date_shift AS (
 SELECT
 customer_unique_id,
 purchase_date,
 LAG (purchase_date) OVER (
 PARTITION BY customer_unique_id
 ORDER BY purchase_date
 ) AS date_shift
 FROM distinct_segment
),
streak_group AS (
 SELECT
 customer_unique_id,
 purchase_date,
 SUM (
 CASE
 WHEN date_shift IS NULL
 OR DATEDIFF (DAY, date_shift, purchase_date) <> 1
 THEN 1
 ELSE 0
 END
 ) OVER (
 PARTITION BY customer_unique_id
 ORDER BY purchase_date
 ) AS streak_id
 FROM date_shift
),
streak_count AS (
 SELECT
 customer_unique_id,
 streak_id,
 COUNT (*) AS streak_length
 FROM streak_group
 GROUP BY customer_unique_id, streak_id
)
SELECT TOP 1
 customer_unique_id,
 streak_length
FROM streak_count
ORDER BY streak_length DESC;
```
